\section{How to Run the Tool}

\subsection{Requirements}
\begin{itemize}
    \item Python 3.8 or higher
    \item pip package manager
\end{itemize}

\subsection{Installation}
\begin{verbatim}
pip install -r requirements.txt
\end{verbatim}

Or install dependencies manually:
\begin{verbatim}
pip install dash plotly pandas numpy pycountry pycountry-convert scikit-learn
\end{verbatim}

\subsection{Running the Application}
\begin{verbatim}
python cia_v1.py
\end{verbatim}

The tool will start at \texttt{http://127.0.0.1:8050}

\subsection{Required Data Files}
Place these CSV files in the \texttt{Datasets/} folder:
\begin{itemize}
    \item \texttt{energyAnalyst.csv} -- Main merged energy dataset (primary)
    \item \texttt{geography\_data.csv} -- Geographic attributes (Area, Coastline, Land Use)
    \item \texttt{government\_and\_civics\_data.csv} -- Government type information
    \item \texttt{energy\_data.csv} -- Raw energy metrics
    \item \texttt{economy\_data.csv} -- Economic indicators (GDP, Trade)
    \item \texttt{demographics\_data.csv} -- Population data
    \item \texttt{transportation\_data.csv} -- Pipeline infrastructure
    \item \texttt{communications\_data.csv} -- Communications metrics
\end{itemize}

\section{Features}

\begin{itemize}
    \item \textbf{Choropleth Map:} Global energy indicator distributions
    \item \textbf{Scatter Plot:} Bivariate relationships with dynamic X/Y axis
    \item \textbf{Bar Chart Rankings:} Top-N countries for any indicator
    \item \textbf{Correlation Heatmap:} Pairwise correlations between indicators
    \item \textbf{Parallel Coordinates:} Multivariate exploration with brushing
    \item \textbf{PCA \& K-Means:} Automated country clustering
    \item \textbf{Bidirectional Highlighting:} Click to highlight across all views
    \item \textbf{Region Filtering:} Filter by continent
\end{itemize}
